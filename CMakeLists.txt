cmake_minimum_required(VERSION 3.20)
project(ForeLexer VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  enable_language(RC)
  add_library(ForeLexer SHARED src/Plugin.cpp src/ForeLexer.cpp src/ForeDetect.cpp src/AutoDetect.cpp src/ThemePalette.cpp src/ForeLexer.rc src/ForeLexer.def)
  target_include_directories(ForeLexer PRIVATE src vendor/npp-sdk)
  target_link_libraries(ForeLexer PRIVATE user32 comctl32)
  set_target_properties(ForeLexer PROPERTIES PREFIX "")
endif()
add_executable(ForeCoreTests tests/core.cpp src/ForeLexer.cpp)
target_include_directories(ForeCoreTests PRIVATE src)
enable_testing()
add_test(NAME ForeCore COMMAND ForeCoreTests)
add_executable(ForeDetectionTests tests/detection.cpp src/ForeDetect.cpp src/ForeLexer.cpp)
target_include_directories(ForeDetectionTests PRIVATE src)
add_test(NAME ForeDetection COMMAND ForeDetectionTests)
